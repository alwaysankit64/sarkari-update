async function loadJobs() { const container = document.getElementById("jobsList"); if (!container) return; try { const { data, error } = await supabaseClient .from("jobs") .select("*") .order("id", { ascending: false }) .limit(100); if (error) { console.error("Jobs Error:", error); container.innerHTML = `

नौकरी की जानकारी लोड नहीं हो सकी

`; return; } if (!data || data.length === 0) { container.innerHTML = `

अभी कोई नई नौकरी उपलब्ध नहीं है

`; return; } container.innerHTML = data.map(job => { return `
${job.category || "Government Job"}

${job.title || ""}

Organization: ${job.organization || ""}

View Details
`; }).join(""); } catch (error) { console.error( "Jobs Database Error:", error ); } }
